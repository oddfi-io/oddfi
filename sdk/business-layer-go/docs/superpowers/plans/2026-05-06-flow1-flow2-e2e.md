> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Wrap every contract method used by the SDK's `flow1_test.go` / `flow2_test.go` as a `business.Service` method (admin writes only — user-side actions stay as test helpers per the design), then run both flows end-to-end against BSC testnet through this library.

**Architecture:** Four new admin writes (`OpenMarket`, `CloseMarket`, `LockParlay`, `Settle`) under `writes/`, mirroring the existing `LockBet` / `CreditDeposit` pattern (chain_actions pipeline + signer_lease + Watcher). `Settle` is structurally different — it goes through the abigen binding directly because the SDK admin client doesn't expose `settle` (gated `onlyMultisig`). Two new e2e tests under build tag `e2e` exercise the full pipeline. User-side `USDT.transfer` and `BettingPool.withdraw` are unexported test helpers in `e2e_helpers_test.go`.

**Tech Stack:** Go 1.22+, gorm/MySQL via testcontainers, `oddfi-contracts-sdk` v0.1.0, abigen bindings, ethclient, BSC testnet.

```go
package writes

import (
	"context"

	"github.com/ethereum/go-ethereum/core/types"

	"github.com/Crypto-Barrett/contracts-business-layer/coretypes"

	sdkadmin "github.com/Crypto-Barrett/oddfi-contracts-sdk/sdk/admin"
)

// OpenMarketReq opens a (matchID, market, line) bucket for new bets. Reverts
// on contract if already open or if max-markets-per-match is exceeded.
type OpenMarketReq struct {
	MatchID        [32]byte
	Market         coretypes.MarketKind
	Line           uint8
	IdempotencyKey string
}

// OpenMarketService is the per-op handle.
type OpenMarketService struct {
	Admin *sdkadmin.Client
	Deps  SubmitDeps
}

// NewOpenMarketService constructs an OpenMarketService.
func NewOpenMarketService(admin *sdkadmin.Client, deps SubmitDeps) *OpenMarketService {
	return &OpenMarketService{Admin: admin, Deps: deps}
}

func (s *OpenMarketService) sdkArgs(req OpenMarketReq) sdkadmin.OpenMarketArgs {
	return sdkadmin.OpenMarketArgs{
		MatchID: req.MatchID,
		Market:  sdkadmin.MarketType(req.Market),
		Line:    req.Line,
	}
}

// Simulate runs an eth_call dry-run.
func (s *OpenMarketService) Simulate(ctx context.Context, req OpenMarketReq) error {
	if err := s.Admin.SimulateOpenMarket(ctx, s.sdkArgs(req)); err != nil {
		return &coretypes.ErrSimulationFailed{Method: "openMarket", Cause: err}
	}
	return nil
}

// Submit broadcasts the tx and returns immediately. Watcher tracks the rest.
func (s *OpenMarketService) Submit(ctx context.Context, req OpenMarketReq) (coretypes.TxHash, coretypes.ActionID, error) {
	args := s.sdkArgs(req)
	return SubmitPipeline(ctx, s.Deps, SubmitInput{
		Op:             "openMarket",
		IdempotencyKey: req.IdempotencyKey,
		Args:           req,
		Simulate: func(ctx context.Context) error {
			return s.Admin.SimulateOpenMarket(ctx, args)
		},
		Broadcast: func(ctx context.Context, _ uint64) (*types.Transaction, error) {
			return s.Admin.OpenMarket(ctx, args)
		},
	})
}

// OpenMarketAndWait broadcasts and blocks until terminal (or ctx).
func (s *OpenMarketService) OpenMarketAndWait(ctx context.Context, req OpenMarketReq, opts ...coretypes.WaitOption) (coretypes.Result, error) {
	_, id, err := s.Submit(ctx, req)
	if err != nil {
		return coretypes.Result{}, err
	}
	return WaitTerminal(ctx, s.Deps.Repo, id, opts...)
}
```

```go
// LockParlay encodes per-leg odds the same way LockBet encodes its single
// LockedOdds: human-form decimal × 1000, truncated to integer. The contract
// then computes combinedOdds = Π(legOdds) / 1000^(n-1).
func TestLockParlay_LegOddsEncoding(t *testing.T) {
	for _, tc := range []struct {
		human   string
		encoded int64
	}{
		{"1.5", 1500},
		{"1.818", 1818},
		{"2.5", 2500},
		{"1.05", 1050},
	} {
		odds, err := decimal.NewFromString(tc.human)
		require.NoError(t, err)
		got := odds.Mul(decimal.NewFromInt(1000)).Truncate(0).BigInt()
		require.Equal(t, big.NewInt(tc.encoded).String(), got.String(), "human %s", tc.human)
	}
}
```

```go
package writes

import (
	"context"
	"errors"
	"fmt"
	"math/big"

	"github.com/ethereum/go-ethereum/common"
	"github.com/ethereum/go-ethereum/core/types"
	"github.com/shopspring/decimal"

	"github.com/Crypto-Barrett/contracts-business-layer/coretypes"
	"github.com/Crypto-Barrett/contracts-business-layer/internal/money"

	sdkadmin "github.com/Crypto-Barrett/oddfi-contracts-sdk/sdk/admin"
)

// LockParlayReq is the low-level input for a multi-leg combo bet. All leg
// slices must be the same length and ≥ 2 (single-leg "parlay" = LockBet).
//
// Stake is human form; library scales by token Decimals.
// LegOdds entries are human form (e.g. 1.5); library multiplies each by 1000.
type LockParlayReq struct {
	ParlayID       [32]byte
	User           common.Address
	Stake          decimal.Decimal
	Token          coretypes.TokenSymbol
	MatchIDs       [][32]byte
	Markets        []coretypes.MarketKind
	Outcomes       []uint8
	Lines          []uint8
	LegOdds        []decimal.Decimal
	LegBetIDs      [][32]byte
	IdempotencyKey string
}

// LockParlayService is the per-op handle.
type LockParlayService struct {
	Admin  *sdkadmin.Client
	Deps   SubmitDeps
	Tokens map[coretypes.TokenSymbol]coretypes.TokenInfo
}

// NewLockParlayService constructs a LockParlayService.
func NewLockParlayService(admin *sdkadmin.Client, deps SubmitDeps, tokens map[coretypes.TokenSymbol]coretypes.TokenInfo) *LockParlayService {
	return &LockParlayService{Admin: admin, Deps: deps, Tokens: tokens}
}

func (s *LockParlayService) sdkArgs(req LockParlayReq) (sdkadmin.LockParlayArgs, error) {
	if err := req.validate(); err != nil {
		return sdkadmin.LockParlayArgs{}, err
	}
	tok, ok := s.Tokens[req.Token]
	if !ok {
		return sdkadmin.LockParlayArgs{}, &coretypes.ErrUnknownToken{Symbol: req.Token}
	}
	markets := make([]sdkadmin.MarketType, len(req.Markets))
	for i, m := range req.Markets {
		markets[i] = sdkadmin.MarketType(m)
	}
	legOdds := make([]*big.Int, len(req.LegOdds))
	for i, o := range req.LegOdds {
		legOdds[i] = o.Mul(decimal.NewFromInt(1000)).Truncate(0).BigInt()
	}
	return sdkadmin.LockParlayArgs{
		ParlayID:  req.ParlayID,
		User:      req.User,
		Stake:     money.ScaleToBase(req.Stake, tok.Decimals),
		MatchIDs:  req.MatchIDs,
		Markets:   markets,
		Outcomes:  req.Outcomes,
		Lines:     req.Lines,
		LegOdds:   legOdds,
		LegBetIDs: req.LegBetIDs,
	}, nil
}

func (req LockParlayReq) validate() error {
	n := len(req.MatchIDs)
	if n < 2 {
		return errors.New("writes: LockParlay requires at least 2 legs")
	}
	if got := len(req.Markets); got != n {
		return fmt.Errorf("writes: LockParlay len(Markets)=%d != len(MatchIDs)=%d", got, n)
	}
	if got := len(req.Outcomes); got != n {
		return fmt.Errorf("writes: LockParlay len(Outcomes)=%d != len(MatchIDs)=%d", got, n)
	}
	if got := len(req.Lines); got != n {
		return fmt.Errorf("writes: LockParlay len(Lines)=%d != len(MatchIDs)=%d", got, n)
	}
	if got := len(req.LegOdds); got != n {
		return fmt.Errorf("writes: LockParlay len(LegOdds)=%d != len(MatchIDs)=%d", got, n)
	}
	if got := len(req.LegBetIDs); got != n {
		return fmt.Errorf("writes: LockParlay len(LegBetIDs)=%d != len(MatchIDs)=%d", got, n)
	}
	return nil
}

// Simulate runs an eth_call dry-run.
func (s *LockParlayService) Simulate(ctx context.Context, req LockParlayReq) error {
	args, err := s.sdkArgs(req)
	if err != nil {
		return err
	}
	if err := s.Admin.SimulateLockParlay(ctx, args); err != nil {
		return &coretypes.ErrSimulationFailed{Method: "lockParlay", Cause: err}
	}
	return nil
}

// Submit broadcasts the tx and returns immediately. Watcher tracks the rest.
func (s *LockParlayService) Submit(ctx context.Context, req LockParlayReq) (coretypes.TxHash, coretypes.ActionID, error) {
	args, err := s.sdkArgs(req)
	if err != nil {
		return "", 0, err
	}
	return SubmitPipeline(ctx, s.Deps, SubmitInput{
		Op:             "lockParlay",
		IdempotencyKey: req.IdempotencyKey,
		Args:           req,
		Simulate: func(ctx context.Context) error {
			return s.Admin.SimulateLockParlay(ctx, args)
		},
		Broadcast: func(ctx context.Context, _ uint64) (*types.Transaction, error) {
			return s.Admin.LockParlay(ctx, args)
		},
	})
}

// LockParlayAndWait broadcasts and blocks until terminal (or ctx).
func (s *LockParlayService) LockParlayAndWait(ctx context.Context, req LockParlayReq, opts ...coretypes.WaitOption) (coretypes.Result, error) {
	_, id, err := s.Submit(ctx, req)
	if err != nil {
		return coretypes.Result{}, err
	}
	return WaitTerminal(ctx, s.Deps.Repo, id, opts...)
}
```

```go
package writes

import (
	"context"
	"fmt"
	"math/big"
	"strings"

	"github.com/ethereum/go-ethereum"
	"github.com/ethereum/go-ethereum/accounts/abi"
	"github.com/ethereum/go-ethereum/accounts/abi/bind"
	"github.com/ethereum/go-ethereum/common"
	"github.com/ethereum/go-ethereum/core/types"
	"github.com/ethereum/go-ethereum/ethclient"

	"github.com/Crypto-Barrett/contracts-business-layer/coretypes"

	"github.com/Crypto-Barrett/oddfi-contracts-sdk/sdk/bindings/bettingpool"
	sdksigner "github.com/Crypto-Barrett/oddfi-contracts-sdk/sdk/signer"
)

// SettleReq settles a market: distributes payouts to winning bets and clears
// locked balances. The contract's `settle` function is `onlyMultisig`. On
// testnet the multisig EOA equals the admin EOA, so this service signs with
// the admin signer. Mainnet (real Gnosis Safe) needs the propose-via-Safe
// path (deferred to a separate effort).
type SettleReq struct {
	MatchID        [32]byte
	Market         coretypes.MarketKind
	Line           uint8
	WinningOutcome uint8
	BetIDs         [][32]byte
	IdempotencyKey string
}

// SettleService is the per-op handle.
type SettleService struct {
	Eth    *ethclient.Client
	Signer sdksigner.Signer
	BPAddr common.Address
	Deps   SubmitDeps
	bp     *bettingpool.BettingPool
	abi    abi.ABI
}

// NewSettleService constructs a SettleService. Validates the binding can be
// built and the ABI parsed (errors here are bugs in caller wiring).
func NewSettleService(eth *ethclient.Client, signer sdksigner.Signer, bpAddr common.Address, deps SubmitDeps) (*SettleService, error) {
	bp, err := bettingpool.NewBettingPool(bpAddr, eth)
	if err != nil {
		return nil, fmt.Errorf("settle: NewBettingPool: %w", err)
	}
	parsed, err := abi.JSON(strings.NewReader(bettingpool.BettingPoolMetaData.ABI))
	if err != nil {
		return nil, fmt.Errorf("settle: parse ABI: %w", err)
	}
	return &SettleService{
		Eth:    eth,
		Signer: signer,
		BPAddr: bpAddr,
		Deps:   deps,
		bp:     bp,
		abi:    parsed,
	}, nil
}

// Simulate runs an eth_call dry-run with msg.sender = admin signer address.
// Catches `not open` / `invalid bet` / wrong-signer reverts before broadcast.
func (s *SettleService) Simulate(ctx context.Context, req SettleReq) error {
	data, err := s.abi.Pack("settle", req.MatchID, uint8(req.Market), req.Line, req.WinningOutcome, req.BetIDs)
	if err != nil {
		return fmt.Errorf("settle: pack calldata: %w", err)
	}
	from := s.Signer.Address()
	if _, err := s.Eth.CallContract(ctx, ethereum.CallMsg{
		From: from,
		To:   &s.BPAddr,
		Data: data,
	}, nil); err != nil {
		return &coretypes.ErrSimulationFailed{Method: "settle", Cause: err}
	}
	return nil
}

// Submit broadcasts the tx and returns immediately. Watcher tracks the rest.
func (s *SettleService) Submit(ctx context.Context, req SettleReq) (coretypes.TxHash, coretypes.ActionID, error) {
	return SubmitPipeline(ctx, s.Deps, SubmitInput{
		Op:             "settle",
		IdempotencyKey: req.IdempotencyKey,
		Args:           req,
		Simulate: func(ctx context.Context) error {
			data, err := s.abi.Pack("settle", req.MatchID, uint8(req.Market), req.Line, req.WinningOutcome, req.BetIDs)
			if err != nil {
				return fmt.Errorf("settle: pack calldata: %w", err)
			}
			_, cerr := s.Eth.CallContract(ctx, ethereum.CallMsg{
				From: s.Signer.Address(),
				To:   &s.BPAddr,
				Data: data,
			}, nil)
			return cerr
		},
		Broadcast: func(ctx context.Context, _ uint64) (*types.Transaction, error) {
			opts, err := s.txOpts(ctx)
			if err != nil {
				return nil, err
			}
			return s.bp.Settle(opts, req.MatchID, uint8(req.Market), req.Line, req.WinningOutcome, req.BetIDs)
		},
	})
}

// SettleAndWait broadcasts and blocks until terminal (or ctx).
func (s *SettleService) SettleAndWait(ctx context.Context, req SettleReq, opts ...coretypes.WaitOption) (coretypes.Result, error) {
	_, id, err := s.Submit(ctx, req)
	if err != nil {
		return coretypes.Result{}, err
	}
	return WaitTerminal(ctx, s.Deps.Repo, id, opts...)
}

// txOpts builds bind.TransactOpts driven by the admin signer. Pulls pending
// nonce from RPC; relies on caller (SubmitPipeline → signer_lease) to hold
// the lock so this query is serialized.
func (s *SettleService) txOpts(ctx context.Context) (*bind.TransactOpts, error) {
	from := s.Signer.Address()
	chainID, err := s.Eth.ChainID(ctx)
	if err != nil {
		return nil, &coretypes.ErrRPC{Op: "ChainID", Cause: err}
	}
	nonce, err := s.Eth.PendingNonceAt(ctx, from)
	if err != nil {
		return nil, &coretypes.ErrRPC{Op: "PendingNonceAt", Cause: err}
	}
	return &bind.TransactOpts{
		From:    from,
		Nonce:   new(big.Int).SetUint64(nonce),
		Context: ctx,
		Signer: func(_ common.Address, tx *types.Transaction) (*types.Transaction, error) {
			return s.Signer.SignTx(ctx, tx, chainID)
		},
	}, nil
}
```

```go
// SimulateOpenMarket runs an eth_call dry-run.
func (s *Service) SimulateOpenMarket(ctx context.Context, req writes.OpenMarketReq) error {
	if s.closed || s.wired == nil || s.wired.full == nil {
		return ErrServiceClosed
	}
	return s.wired.full.openMarket.Simulate(ctx, req)
```

```go
// userTxOpts builds bind.TransactOpts using the user signer (parallels SDK
// e2e helpers_test.go:txOpts but kept private to this package).
func userTxOpts(ctx context.Context, conn *sdkclient.Connection, s sdksigner.Signer) (*bind.TransactOpts, error) {
	from := s.Address()
	chainID := conn.ChainID()
	nonce, err := conn.EthClient().PendingNonceAt(ctx, from)
	if err != nil {
		return nil, fmt.Errorf("pending nonce: %w", err)
	}
	return &bind.TransactOpts{
		From:    from,
		Nonce:   new(big.Int).SetUint64(nonce),
		Context: ctx,
		Signer: func(_ common.Address, tx *types.Transaction) (*types.Transaction, error) {
			return s.SignTx(ctx, tx, chainID)
		},
	}, nil
}

// rand32 returns a random 32-byte id (suitable for matchID/betID/parlayID).

	t.Helper()

	_, err := rand.Read(out[:])
	require.NoError(t, err)
	return out
}

// usdt6 formats a *big.Int as USDT 6 decimals.
func usdt6(v *big.Int) string {
	if v == nil {
		return "<nil>"
	}
	abs := new(big.Int).Abs(v)
	whole, frac := new(big.Int).QuoRem(abs, big.NewInt(1_000_000), new(big.Int))
	sign := ""
	if v.Sign() < 0 {
		sign = "-"
	}
	return fmt.Sprintf("%s%s.%06d", sign, whole.String(), frac.Int64())
}

// signed wraps usdt6 and prefixes positives with '+'.
func signed(v *big.Int) string {
	s := usdt6(v)
	if v.Sign() > 0 {
		return "+" + s
	}
	return s
}

// stripHexPrefix removes a leading "0x"/"0X" if present.
func stripHexPrefix(h string) string {
	if len(h) >= 2 && (h[:2] == "0x" || strings.EqualFold(h[:2], "0X")) {
		return h[2:]
	}
	return h
}

// waitOpts are the WaitOption set we use across flow steps (poll fast, the
// flow can wait long enough for testnet finality at the configured
// confirmations level).
func waitOpts() []coretypes.WaitOption {
	return []coretypes.WaitOption{coretypes.WaitPollInterval(2 * time.Second)}
}
```

```go
	t.Logf("    Settle confirmed tx=%s", settleRes.TxHash)

	lockedAfterSettle, err := e.bpCall.LockedBalances(nil, user)
	require.NoError(t, err)
	require.Equal(t, 0, lockedAfterSettle.Sign(),
		"after Settle user.lockedBalances = %s, want 0", usdt6(lockedAfterSettle))

	t.Log("STEP 5  user.Withdraw (full payout)")

	_ = userWithdraw(t, e, payout)

	require.NoError(t, err)

	require.NoError(t, err)
```

```go
//go:build e2e

package business_test

// TestFlow2_E2E mirrors sdk/e2e/flow2_test.go (parlay): user stakes 1 USDT
// across 2 legs at 1.5x each → combined 2.25x, both legs win.
//
// Step list:
//
//	0a/0b  admin   SubmitOpenMarket × 2
//	1      user    userTransferUSDT
//	2      admin   SubmitCreditDeposit
//	3      admin   SubmitLockParlay
//	3.5a/b admin   SubmitCloseMarket × 2
//	4a/b   admin   SubmitSettle × 2 (final leg auto-finalises parlay)
//	5      user    userWithdraw
//
// Run:
//
//	E2E_RPC=... E2E_ADMIN_PK=... E2E_USER_PK=... \
//	  go test -tags=e2e -count=1 -v -timeout=15m -run TestFlow2_E2E ./...

import (
	"math/big"
	"testing"
	"time"

	"github.com/shopspring/decimal"
	"github.com/stretchr/testify/require"

	"github.com/Crypto-Barrett/contracts-business-layer/coretypes"
	"github.com/Crypto-Barrett/contracts-business-layer/writes"
)

// Verbatim from sdk/e2e/flow2_test.go.
const (
	f2Stake     int64 = 1_000_000 // 1 USDT
	f2LegOdds   int64 = 1500      // 1.5x → combined 2 legs = 2250
	f2Market    uint8 = 0         // X12
	f2Line      uint8 = 0
	f2Outcome   uint8 = 1
	f2Winning   uint8 = 1
	f2NumLegs         = 2
)

func TestFlow2_E2E(t *testing.T) {
	e := setupE2E(t, 15*time.Minute)
	user := e.usrS.Address()
	t.Logf("Admin: %s   User: %s", e.admS.Address().Hex(), user.Hex())

	stakeWei := big.NewInt(f2Stake)

	startUserUSDT, err := e.usdt.BalanceOf(nil, user)
	require.NoError(t, err)
	startBpUSDT, err := e.usdt.BalanceOf(nil, e.bpAddr)
	require.NoError(t, err)
	t.Logf("== START ==  user.USDT=%s  bp.USDT=%s", usdt6(startUserUSDT), usdt6(startBpUSDT))
	require.GreaterOrEqual(t, startUserUSDT.Cmp(stakeWei), 0,
		"user wallet has %s USDT < required %s", usdt6(startUserUSDT), usdt6(stakeWei))

	matchIDs := [f2NumLegs][32]byte{}
	legBetIDs := [f2NumLegs][32]byte{}
	for i := 0; i < f2NumLegs; i++ {
		matchIDs[i] = rand32(t)
		legBetIDs[i] = rand32(t)
	}
	parlayID := rand32(t)
	t.Logf("parlayID = 0x%x", parlayID)
	for i := 0; i < f2NumLegs; i++ {
		t.Logf("  leg %d: matchID=0x%x betID=0x%x", i, matchIDs[i], legBetIDs[i])
	}
	idemPrefix := "flow2-" + time.Now().UTC().Format("20060102T150405.000")

	// ── 0a/0b. admin.SubmitOpenMarket × 2 ───────────────────────────────────
	for i := 0; i < f2NumLegs; i++ {
		label := string(rune('a' + byte(i)))
		t.Logf("STEP 0%s admin.SubmitOpenMarket leg=%d", label, i)
		req := writes.OpenMarketReq{
			MatchID:        matchIDs[i],
			Market:         coretypes.MarketKind(f2Market),
			Line:           f2Line,
			IdempotencyKey: idemPrefix + ":openMarket:" + label,
		}
		require.NoError(t, e.svc.SimulateOpenMarket(e.ctx, req), "simulate OpenMarket %d", i)
		res, err := e.svc.OpenMarketAndWait(e.ctx, req, waitOpts()...)
		require.NoError(t, err, "OpenMarketAndWait %d", i)
		require.Equal(t, coretypes.StatusConfirmed, res.Status)
		t.Logf("    confirmed tx=%s block=%d", res.TxHash, res.BlockNumber)
	}

	// ── 1. user → BettingPool USDT.transfer ─────────────────────────────────
	t.Log("STEP 1  user → BettingPool USDT.transfer")
	depositTxHash := userTransferUSDT(t, e, e.bpAddr, stakeWei)

	// ── 2. admin.SubmitCreditDeposit ────────────────────────────────────────
	t.Log("STEP 2  admin.SubmitCreditDeposit")
	depReq := writes.CreditDepositReq{
		User:           user,
		Amount:         decimal.NewFromInt(1),
		Token:          "USDT",
		OffchainTxHash: depositTxHash,
		IdempotencyKey: idemPrefix + ":creditDeposit",
	}
	require.NoError(t, e.svc.SimulateCreditDeposit(e.ctx, depReq))
	depRes, err := e.svc.CreditDepositAndWait(e.ctx, depReq, waitOpts()...)
	require.NoError(t, err)
	require.Equal(t, coretypes.StatusConfirmed, depRes.Status)
	t.Logf("    CreditDeposit confirmed tx=%s", depRes.TxHash)

	// ── 3. admin.SubmitLockParlay ───────────────────────────────────────────
	t.Log("STEP 3  admin.SubmitLockParlay (2 legs, 1.5x each)")
	legOdds := make([]decimal.Decimal, f2NumLegs)
	markets := make([]coretypes.MarketKind, f2NumLegs)
	outcomes := make([]uint8, f2NumLegs)
	lines := make([]uint8, f2NumLegs)
	for i := 0; i < f2NumLegs; i++ {
		legOdds[i] = decimal.NewFromFloat(1.5)
		markets[i] = coretypes.MarketKind(f2Market)
		outcomes[i] = f2Outcome
		lines[i] = f2Line
	}
	parlayReq := writes.LockParlayReq{
		ParlayID:       parlayID,
		User:           user,
		Stake:          decimal.NewFromInt(1),
		Token:          "USDT",
		MatchIDs:       matchIDs[:],
		Markets:        markets,
		Outcomes:       outcomes,
		Lines:          lines,
		LegOdds:        legOdds,
		LegBetIDs:      legBetIDs[:],
		IdempotencyKey: idemPrefix + ":lockParlay",
	}
	require.NoError(t, e.svc.SimulateLockParlay(e.ctx, parlayReq), "simulate LockParlay")
	parlayRes, err := e.svc.LockParlayAndWait(e.ctx, parlayReq, waitOpts()...)
	require.NoError(t, err, "LockParlayAndWait")
	require.Equal(t, coretypes.StatusConfirmed, parlayRes.Status)
	t.Logf("    LockParlay confirmed tx=%s", parlayRes.TxHash)

	parlay, err := e.bpCall.Parlays(nil, parlayID)
	require.NoError(t, err)
	t.Logf("    parlay user=%s stake=%s combined=%s maxPayout=%s status=%d",
		parlay.User.Hex(), usdt6(parlay.Stake), parlay.CombinedOdds.String(),
		usdt6(parlay.MaxPayout), parlay.Status)
	require.Equal(t, user, parlay.User)
	require.Equal(t, 0, parlay.Stake.Cmp(stakeWei))
	require.Equal(t, big.NewInt(2250).String(), parlay.CombinedOdds.String(),
		"combinedOdds 1.5×1.5 = 2.25x → 2250 (×1000 encoded)")

	locked, err := e.bpCall.LockedBalances(nil, user)
	require.NoError(t, err)
	require.Equal(t, 0, locked.Cmp(stakeWei),
		"after LockParlay user.lockedBalances = %s, want %s", usdt6(locked), usdt6(stakeWei))

	// ── 3.5a/b. admin.SubmitCloseMarket × 2 ─────────────────────────────────
	for i := 0; i < f2NumLegs; i++ {
		label := string(rune('a' + byte(i)))
		t.Logf("STEP 3.5%s admin.SubmitCloseMarket leg=%d", label, i)
		req := writes.CloseMarketReq{
			MatchID:        matchIDs[i],
			Market:         coretypes.MarketKind(f2Market),
			Line:           f2Line,
			IdempotencyKey: idemPrefix + ":closeMarket:" + label,
		}
		require.NoError(t, e.svc.SimulateCloseMarket(e.ctx, req), "simulate CloseMarket %d", i)
		res, err := e.svc.CloseMarketAndWait(e.ctx, req, waitOpts()...)
		require.NoError(t, err, "CloseMarketAndWait %d", i)
		require.Equal(t, coretypes.StatusConfirmed, res.Status)
	}

	// ── 4a/b. admin.SubmitSettle × 2 ────────────────────────────────────────
	for i := 0; i < f2NumLegs; i++ {
		label := string(rune('a' + byte(i)))
		t.Logf("STEP 4%s admin.SubmitSettle leg=%d", label, i)
		req := writes.SettleReq{
			MatchID:        matchIDs[i],
			Market:         coretypes.MarketKind(f2Market),
			Line:           f2Line,
			WinningOutcome: f2Winning,
			BetIDs:         [][32]byte{legBetIDs[i]},
			IdempotencyKey: idemPrefix + ":settle:" + label,
		}
		require.NoError(t, e.svc.SimulateSettle(e.ctx, req), "simulate Settle %d", i)
		res, err := e.svc.SettleAndWait(e.ctx, req, waitOpts()...)
		require.NoError(t, err, "SettleAndWait %d", i)
		require.Equal(t, coretypes.StatusConfirmed, res.Status)

		parlay, _ := e.bpCall.Parlays(nil, parlayID)
		t.Logf("    Settle leg %d confirmed; parlay status=%d settled=%d won=%d",
			i, parlay.Status, parlay.SettledLegs, parlay.WonLegs)
	}

	// post-settle assertions
	finalLocked, err := e.bpCall.LockedBalances(nil, user)
	require.NoError(t, err)
	require.Equal(t, 0, finalLocked.Sign(),
		"after all legs settled user.lockedBalances = %s, want 0", usdt6(finalLocked))
	finalBal, err := e.bpCall.UserBalances(nil, user)
	require.NoError(t, err)
	require.Greater(t, finalBal.Cmp(stakeWei), 0,
		"after all legs settled user.userBalances = %s, want > stake %s",
		usdt6(finalBal), usdt6(stakeWei))

	parlay, _ = e.bpCall.Parlays(nil, parlayID)
	wantPayout := new(big.Int).Mul(stakeWei, parlay.CombinedOdds)
	wantPayout.Quo(wantPayout, big.NewInt(1000))
	require.Equal(t, 0, finalBal.Cmp(wantPayout),
		"user.userBalances = %s, want %s (stake %s × combined %s / 1000)",
		usdt6(finalBal), usdt6(wantPayout), usdt6(stakeWei), parlay.CombinedOdds)
	t.Logf("    payout = %s × %s / 1000 = %s ✓", usdt6(stakeWei), parlay.CombinedOdds,
		usdt6(wantPayout))

	// ── 5. user.Withdraw ────────────────────────────────────────────────────
	t.Log("STEP 5  user.Withdraw (full payout)")
	payout := finalBal
	_ = userWithdraw(t, e, payout)

	endUserUSDT, err := e.usdt.BalanceOf(nil, user)
	require.NoError(t, err)
	endBal, err := e.bpCall.UserBalances(nil, user)
	require.NoError(t, err)
	require.Equal(t, 0, endBal.Sign())

	walletDelta := new(big.Int).Sub(endUserUSDT, startUserUSDT)
	wantDelta := new(big.Int).Sub(payout, stakeWei)
	require.Equal(t, 0, walletDelta.Cmp(wantDelta),
		"wallet delta = %s, want %s", signed(walletDelta), signed(wantDelta))

	t.Log("== DONE ==")
	t.Logf("stake placed         : -%s USDT", usdt6(stakeWei))
	t.Logf("parlay payout (gross): +%s USDT", usdt6(payout))
	t.Logf("net to user wallet   : %s USDT", signed(walletDelta))
}
```
